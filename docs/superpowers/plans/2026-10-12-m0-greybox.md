# Diwan M0 (grey box) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show every agent behaviour from the spec's grammar in a plain 3D grey-box House of Wisdom, driven both by live Claude Code sessions and by a scripted demo, without changing upstream Pixel Agents code beyond one line.

**Architecture:**

- A new workspace `webview-3d/` (Vite + React 19 + React Three Fiber) sits beside upstream's `webview-ui/`. It reuses upstream's engine (`OfficeState`) and message hook (`useExtensionMessages`) unchanged.
- A Vite plugin redirects upstream's singleton `transport` import to our own transport. Ours picks live or demo, swaps in our world's floor plan, and keeps the client read-only.
- In live mode, Vite's dev server proxies `/ws` to the already-running `pixel-agents` server.

**Tech Stack:** TypeScript 5.9 (strict, `erasableSyntaxOnly`), React 19.2, three 0.185, @react-three/fiber 9.7, @react-three/drei 10.7, Vite 8, Vitest 4, Playwright (root devDependency).

**Spec:** `docs/superpowers/specs/2026-10-12-history-worlds-design.md` (milestone M0). Code map with line references: `docs/architecture-notes.md`.

## Review Focus

1. **Seat ids from the server.** The live server sends `existingAgents` with seat ids from the pixel-agents layout, which don't exist in our world. Agents must still get seats in our world. Pinned in Task 4 (`existingAgents` loses `seatId`).
2. **Server not running.** If the pixel-agents server is down, live mode must show a disconnected state, not crash or render an empty error. Pinned in Task 4 (`FilteredTransport` forwards state changes) and Task 6 (HUD shows connection state).
3. **More sessions than seats.** With more agents than the 12 seats, the extras must still appear, standing, without errors. Pinned in Task 3 (14 agents in a 12-seat world).
4. **Demo loop restart.** Each demo loop must not duplicate agents. Pinned in Task 5 (every created agent is closed before the loop ends) and Task 4 (the loop replays the steps once per loop, even after repeated `webviewReady`).
5. **Late furniture message.** If the server sends `furnitureAssetsLoaded` after our world catalog is installed, it must not replace our catalog and wipe our seats. Pinned in Task 4 (the message is dropped).

```ts
// webview-3d/src/grammar/visualState.ts
import type { Character } from '../../../webview-ui/src/office/types.js';
import { CharacterState } from '../../../webview-ui/src/office/types.js';
import type { ToolKind } from './toolKind.js';
import { toolKind } from './toolKind.js';

export type Pose = 'walk' | 'work' | 'idle';

export interface VisualState {
  pose: Pose;
  toolKind: ToolKind | null;
  signal: 'permission' | 'waiting' | null;
  isSubagent: boolean;
  effect: 'spawn' | 'despawn' | null;
  contextRatio: number | null;
}

export function contextRatio(
  ch: Pick<Character, 'contextTokens' | 'maxContextTokens'>,
): number | null {
  // Zero usage is "no news" upstream (contextUsage.ts): no gauge until real data arrives.
  if (!ch.maxContextTokens || ch.maxContextTokens <= 0 || ch.contextTokens <= 0) return null;
  return Math.min(1, Math.max(0, ch.contextTokens / ch.maxContextTokens));
}

/** Maps engine state to what every world must show (spec §4). */
export function toVisualState(ch: Character): VisualState {
  const pose: Pose =
    ch.state === CharacterState.WALK ? 'walk' : ch.state === CharacterState.TYPE ? 'work' : 'idle';
  return {
    pose,
    toolKind: ch.isActive ? toolKind(ch.currentTool) : null,
    signal: ch.bubbleType,
    isSubagent: ch.isSubagent,
    effect: ch.matrixEffect,
    contextRatio: contextRatio(ch),
  };
}

/** Stable key: React state only updates when something visible changes. */
export function visualSignature(v: VisualState): string {
  const gauge = v.contextRatio === null ? '-' : String(Math.round(v.contextRatio * 20));
  return [v.pose, v.toolKind ?? '-', v.signal ?? '-', v.isSubagent, v.effect ?? '-', gauge].join(
    '|',
  );
}

export function gaugeColor(ratio: number): string {
  if (ratio < 0.6) return '#3da35d';
  if (ratio < 0.85) return '#e0a458';
  return '#d64545';
}

export function countCharacters(chars: Character[]): {
  agents: number;
  subagents: number;
  needsYou: number;
} {
  let agents = 0;
  let subagents = 0;
  let needsYou = 0;
  for (const ch of chars) {
    if (ch.isGreeter) continue;
    if (ch.isSubagent) subagents++;
    else agents++;
    if (ch.bubbleType === 'permission') needsYou++;
  }
  return { agents, subagents, needsYou };
}
```

```ts
// webview-3d/src/scene/coords.ts
import { TILE_SIZE } from '../../../webview-ui/src/constants.js';
import { Direction } from '../../../webview-ui/src/office/types.js';

export type Vec3 = [number, number, number];

/** Engine pixels (tile centre = col*16+8) → world units (1 unit = 1 tile, Y up). */
export function pixelToWorld(x: number, y: number, height = 0): Vec3 {
  return [x / TILE_SIZE, height, y / TILE_SIZE];
}

export function footprintCenter(col: number, row: number, w: number, h: number): Vec3 {
  return [col + w / 2, 0, row + h / 2];
}

const ROTATION_BY_DIRECTION: Record<Direction, number> = {
  [Direction.DOWN]: 0,
  [Direction.RIGHT]: Math.PI / 2,
  [Direction.UP]: Math.PI,
```

```ts
export function resolveMode(search: string, envDemo: string | undefined): TransportMode {
  const params = new URLSearchParams(search);
  if (params.has('demo')) return params.get('demo') === '0' ? 'live' : 'demo';
```

```ts
  readonly ready: Promise<void> = Promise.resolve();

    if (message.type !== 'webviewReady') return;
    // Like the real server: every webviewReady gets the current state again.
    for (const msg of this.timeline.setup) this.emit(msg);
    if (this.playing) return;
    this.playing = true;
    this.playLoop();
  }

  onMessage(handler: (message: ServerMessage) => void): () => void {
    this.handlers.add(handler);
    return () => {
      this.handlers.delete(handler);
    };
  }

  onStateChange(_handler: (state: TransportState) => void): () => void {
    return () => {};
  }

  dispose(): void {
    for (const timer of this.timers) clearTimeout(timer);
    this.timers = [];
    this.handlers.clear();
    this.playing = false;
  }

  private emit(message: ServerMessage): void {
    for (const handler of [...this.handlers]) handler(message);
  }

  private playLoop(): void {
    const { speed, loop } = this.options;
    this.timers = this.timeline.steps.map((step) =>
      setTimeout(() => this.emit(step.msg), step.at / speed),
    );
    if (loop) {
      this.timers.push(setTimeout(() => this.playLoop(), this.timeline.durationMs / speed));
    }
  }
}
```

```ts
import type { MessageTransport, TransportState } from '../../../core/src/transport.js';
import type { OfficeLayout } from '../../../webview-ui/src/office/types.js';

// 2D art and the consent tour have no place in the 3D client. Dropping
// furnitureAssetsLoaded also keeps our world catalog from being replaced.
const DROPPED_INCOMING = new Set<string>([
  'furnitureAssetsLoaded',
  'characterSpritesLoaded',
  'petSpritesLoaded',
  'floorTilesLoaded',
  'wallTilesLoaded',
  'carpetTilesLoaded',
  'hooksConsentRequest',
]);

// Read-only client: never writes seats, layout, settings or hooks on the shared server.
const ALLOWED_OUTGOING = new Set<string>(['webviewReady']);

export function filterIncoming(msg: ServerMessage, floorPlan: OfficeLayout): ServerMessage | null {
  if (DROPPED_INCOMING.has(msg.type)) return null;
  if (msg.type === 'layoutLoaded') return { type: 'layoutLoaded', layout: floorPlan };
  if (msg.type === 'existingAgents') {
    const agentMeta = Object.fromEntries(
      Object.entries(msg.agentMeta).map(([id, meta]) => [
        id,
        { palette: meta.palette, hueShift: meta.hueShift },
      ]),
    );
    return { ...msg, agentMeta };
  }
  return msg;
}

export function isAllowedOutgoing(msg: ClientMessage): boolean {
  return ALLOWED_OUTGOING.has(msg.type);
}

export class FilteredTransport implements MessageTransport {
  private readonly inner: MessageTransport;
  private readonly floorPlan: OfficeLayout;

  constructor(inner: MessageTransport, floorPlan: OfficeLayout) {
    this.inner = inner;
    this.floorPlan = floorPlan;
  }

  get ready(): Promise<void> {
    return this.inner.ready;
  }

  get state(): TransportState {
    return this.inner.state;
  }

  send(message: ClientMessage): void {
    if (isAllowedOutgoing(message)) this.inner.send(message);
  }

  onMessage(handler: (message: ServerMessage) => void): () => void {
    return this.inner.onMessage((message) => {
      const filtered = filterIncoming(message, this.floorPlan);
      if (filtered) handler(filtered);
    });
  }

  onStateChange(handler: (state: TransportState) => void): () => void {
    return this.inner.onStateChange(handler);
  }

  dispose(): void {
    this.inner.dispose();
  }
}
```

```ts
/**

export function redirectUpstreamTransport(
  upstreamTransportPath: string,
  ourTransportPath: string,
): Plugin {
  return {
    name: 'diwan-redirect-upstream-transport',
    enforce: 'pre',
    async resolveId(source, importer, options) {
      if (!importer || !source.includes('transport/index')) return null;
      const resolved = await this.resolve(source, importer, { ...options, skipSelf: true });
      return shouldRedirect(resolved?.id, upstreamTransportPath) ? ourTransportPath : null;
    },
  };
```

```ts
export interface TileCoord {
  col: number;
  row: number;
}

export function splitTiles(layout: OfficeLayout): { floor: TileCoord[]; walls: TileCoord[] } {
  const floor: TileCoord[] = [];
  const walls: TileCoord[] = [];
  layout.tiles.forEach((tile, index) => {
    const coord = { col: index % layout.cols, row: Math.floor(index / layout.cols) };
    if (tile === TileType.WALL) walls.push(coord);
    else if (tile !== TileType.VOID) floor.push(coord);
  });
  return { floor, walls };
}
```

```tsx
function PermissionBeacon() {
  const ref = useRef<Mesh>(null);
  useFrame(({ clock }) => {
    if (ref.current) ref.current.scale.setScalar(1 + Math.sin(clock.elapsedTime * 6) * 0.2);
  });
  return (
    <mesh ref={ref} position={[0, 1.7, 0]}>
      <sphereGeometry args={[0.16, 24, 16]} />
      <meshStandardMaterial color="#ffc23d" emissive="#ffb000" emissiveIntensity={1.2} />
    </mesh>
  );
}

function ContextGauge({ ratio }: { ratio: number }) {
  const height = Math.max(ratio * 0.9, 0.02);
  return (
    <mesh position={[0.38, height / 2, 0]}>
      <boxGeometry args={[0.06, height, 0.06]} />
      <meshStandardMaterial
        color={gaugeColor(ratio)}
        emissive={gaugeColor(ratio)}
        emissiveIntensity={0.4}
      />
    </mesh>
  );
}

export function AgentAvatar(props: {
  id: number;
  getOfficeState: () => OfficeState;
  info: AgentInfo;
}) {
  const { id, getOfficeState, info } = props;
  const group = useRef<Group>(null);
  const body = useRef<Mesh>(null);
  const [visual, setVisual] = useState<VisualState | null>(null);
  const signatureRef = useRef('');
  const color = useMemo(() => {
    const palette = getOfficeState().characters.get(id)?.palette ?? 0;
    return PALETTE_COLORS[palette % PALETTE_COLORS.length];
  }, [getOfficeState, id]);

  useFrame(({ clock }) => {
    const ch = getOfficeState().characters.get(id);
    const g = group.current;
    if (!ch || !g) return;
    const [x, , z] = pixelToWorld(ch.x, ch.y);
    g.position.set(x, 0, z);
    g.rotation.y = directionToRotationY(ch.dir);
    const v = toVisualState(ch);
    const t = clock.elapsedTime;
    const bob =
      v.pose === 'walk'
        ? Math.abs(Math.sin(t * 10)) * 0.08
        : v.pose === 'work'
          ? Math.sin(t * 6) * 0.02
          : 0;
    if (body.current) body.current.position.y = BODY_Y + bob;
    const progress = Math.min(1, ch.matrixEffectTimer / MATRIX_EFFECT_DURATION_SEC);
    const effectScale = v.effect === 'spawn' ? progress : v.effect === 'despawn' ? 1 - progress : 1;
    g.scale.setScalar(Math.max((v.isSubagent ? 0.75 : 1) * effectScale, 0.001));
    const signature = visualSignature(v);
    if (signature !== signatureRef.current) {
      signatureRef.current = signature;
      setVisual(v);
    }
  });

  return (
    <group ref={group}>
      <mesh ref={body} position={[0, BODY_Y, 0]} castShadow>
        <capsuleGeometry args={[0.22, 0.5, 8, 16]} />
        <meshStandardMaterial color={color} roughness={0.55} />
      </mesh>
      {/* Facing marker */}
      <mesh position={[0, BODY_Y + 0.15, 0.2]}>
        <boxGeometry args={[0.1, 0.06, 0.08]} />
        <meshStandardMaterial color="#1b1712" />
      </mesh>
      {visual?.toolKind && (
        <mesh position={[0, 1.35, 0]}>
          <boxGeometry args={[0.18, 0.18, 0.18]} />
          <meshStandardMaterial
            color={KIND_COLORS[visual.toolKind]}
            emissive={KIND_COLORS[visual.toolKind]}
            emissiveIntensity={0.6}
          />
        </mesh>
      )}
      {visual?.signal === 'permission' && <PermissionBeacon />}
      {visual?.signal === 'waiting' && (
        <mesh position={[0, 1.6, 0]} rotation={[Math.PI / 2, 0, 0]}>
          <torusGeometry args={[0.18, 0.03, 8, 24]} />
          <meshStandardMaterial color="#ffffff" emissive="#ffffff" emissiveIntensity={0.5} />
        </mesh>
      )}
      {visual?.contextRatio != null && <ContextGauge ratio={visual.contextRatio} />}
      <SpeechBubble bubble={visual ? bubbleContent(visual, info.status) : null} name={info.name} />
    </group>
  );
}
```

```ts
// webview-3d/src/board/feed.ts
import type { ServerMessage } from '../../../core/src/messages.js';
import { toolKind } from '../grammar/toolKind.js';

export interface FeedEntry {
  id: number;
  at: number;
  text: string;
  tone: 'info' | 'approval' | 'done';
}

export interface FeedState {
  entries: FeedEntry[];
  names: Record<number, string>;
  nextId: number;
}

export const EMPTY_FEED: FeedState = { entries: [], names: {}, nextId: 1 };
export const FEED_LIMIT = 30;

const nameOf = (state: FeedState, id: number) => state.names[id] ?? `Agent ${id}`;

function push(state: FeedState, now: number, text: string, tone: FeedEntry['tone']): FeedState {
  if (state.entries[0]?.text === text) return state;
  const entry: FeedEntry = { id: state.nextId, at: now, text, tone };
  return {
    ...state,
    entries: [entry, ...state.entries].slice(0, FEED_LIMIT),
    nextId: state.nextId + 1,
  };
}

/** Notable events only (spec §4): routine reads and edits stay off the board. */
export function feedReducer(state: FeedState, msg: ServerMessage, now: number): FeedState {
  switch (msg.type) {
    case 'existingAgents': {
      const names = { ...state.names };
      for (const [id, folder] of Object.entries(msg.folderNames)) names[Number(id)] = folder;
      return { ...state, names };
    }
    case 'agentCreated': {
      const named = {
        ...state,
        names: { ...state.names, [msg.id]: msg.folderName ?? `Agent ${msg.id}` },
      };
      return push(named, now, `${nameOf(named, msg.id)} joined`, 'info');
    }
    case 'agentToolStart': {
      if (toolKind(msg.toolName) !== 'delegate') return state;
      const task = msg.status.startsWith('Subtask:')
        ? msg.status.slice('Subtask:'.length).trim()
        : msg.status;
      return push(state, now, `${nameOf(state, msg.id)} delegated: ${task}`, 'info');
    }
    case 'agentToolPermission':
      return push(state, now, `${nameOf(state, msg.id)} needs your approval`, 'approval');
    case 'subagentToolPermission':
      return push(
        state,
        now,
        `${nameOf(state, msg.id)}'s subagent needs your approval`,
        'approval',
      );
    case 'agentToolPermissionClear':
      return push(state, now, `${nameOf(state, msg.id)} can continue`, 'info');
    case 'subagentClear':
      return push(state, now, `${nameOf(state, msg.id)}'s subagent returned`, 'done');
    case 'agentStatus':
      if (msg.status !== 'waiting') return state;
      return push(
        state,
        now,
        msg.awaitingInput
          ? `${nameOf(state, msg.id)} is waiting for you`
          : `${nameOf(state, msg.id)} finished`,
        'done',
      );
    case 'agentClosed': {
      const left = push(state, now, `${nameOf(state, msg.id)} left`, 'info');
      const names = { ...left.names };
      delete names[msg.id];
      return { ...left, names };
    }
    default:
      return state;
  }
}
```

```ts
// webview-3d/src/board/useMessageFeed.ts
import { useEffect, useState } from 'react';

import { transport } from '../transport/index.js';
import type { FeedEntry } from './feed.js';
import { EMPTY_FEED, feedReducer } from './feed.js';

export function useMessageFeed(): FeedEntry[] {
  const [feed, setFeed] = useState(EMPTY_FEED);
  useEffect(
    () => transport.onMessage((msg) => setFeed((state) => feedReducer(state, msg, Date.now()))),
    [],
  );
  return feed.entries;
}
```

```ts
test('demo build renders the world and plays the agent grammar', async ({ page }) => {
```
